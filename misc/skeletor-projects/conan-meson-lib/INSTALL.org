* Debug
** Install deps
#+begin_src shell
  conan install . --output-folder=build --build=missing -s build_type=Debug -s compiler.cppstd=20
#+end_src

** Meson
#+begin_src shell
  meson setup --native-file=build/conan_meson_native.ini build
  meson compile -C build
  meson test -C build
#+end_src

* Release
** Install deps
#+begin_src shell
  conan install . --output-folder=build --build=missing -s build_type=Release -s compiler.cppstd=20
#+end_src

** Meson
#+begin_src shell
  meson setup --native-file=build/conan_meson_native.ini build --buildtype=release
  meson compile -C build
  meson test -C build
#+end_src

* Profile Example (Windows / MSVC)
#+begin_src conf
  [settings]
  arch=x86_64
  build_type=Release
  compiler=msvc
  compiler.cppstd=20
  compiler.runtime=dynamic
  compiler.version=193
  os=Windows
#+end_src
